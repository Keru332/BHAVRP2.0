# Leer datos
costos <- read.csv("tabla costos ruta normalizadas.csv",
                   row.names = 1,
                   check.names = FALSE)

metricas <- read.csv("tabla metricas normalizadas.csv",
                     row.names = 1,
                     check.names = FALSE)

# Alinear filas por nombre
filas_comunes <- intersect(rownames(costos), rownames(metricas))

if (length(filas_comunes) == 0) {
  stop("No hay nombres de fila en común entre ambas tablas.")
}

costos <- costos[filas_comunes, , drop = FALSE]
metricas <- metricas[filas_comunes, , drop = FALSE]

# Carpeta de salida
out_dir <- "correlaciones_heuristica_costo"
dir.create(out_dir, showWarnings = FALSE)

# Recorrer cada heurística de costo de ruta
for (heur in colnames(costos)) {
 
  resultado <- data.frame(
    Metrica = colnames(metricas),
    Pearson = NA_real_,
    Pearson_pvalue = NA_real_,
    Spearman = NA_real_,
    Spearman_pvalue = NA_real_,
    stringsAsFactors = FALSE
  )
 
  for (i in seq_along(colnames(metricas))) {
    metrica <- colnames(metricas)[i]
   
    x <- costos[[heur]]
    y <- metricas[[metrica]]
   
    ok <- complete.cases(x, y)
    x <- x[ok]
    y <- y[ok]
   
    resultado$Pearson[i] <- cor(x[ok], y[ok], method = "pearson")
    resultado$Spearman[i] <- cor(x[ok], y[ok], method = "spearman")
  }
 
  # Nombre de archivo seguro
  nombre_archivo <- paste0(
    "correlaciones_",
    gsub("[^A-Za-z0-9_-]", "_", heur),
    ".csv"
  )
 
  write.csv(
    resultado,
    file.path(out_dir, nombre_archivo),
    row.names = FALSE
  )
}

cat("Listo. Archivos generados en:", out_dir, "\n")
